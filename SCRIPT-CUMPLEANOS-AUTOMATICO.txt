function enviarRecordatoriosCumpleanos() {
  try {
    // 📧 CAMBIA ESTE EMAIL POR EL TUYO
    var emailDestino = "tu-email@gmail.com"; // ← CAMBIA ESTO
    
    var sheet = SpreadsheetApp.getActiveSpreadsheet().getActiveSheet();
    var data = sheet.getDataRange().getValues();
    
    // Fecha de hoy
    var hoy = new Date();
    var diaHoy = hoy.getDate();
    var mesHoy = hoy.getMonth() + 1;
    
    // Arrays para almacenar cumpleaños
    var cumpleañosHoy = [];
    var cumpleañosProximos3Dias = [];
    var cumpleañosProximos7Dias = [];
    
    // Revisar cada fila (saltando el header en fila 0)
    for (var i = 1; i < data.length; i++) {
      var row = data[i];
      
      // Columnas del Club de Amigos:
      // 0: Fecha de registro
      // 1: Nombre
      // 2: RUT
      // 3: Teléfono
      // 4: Fecha de Nacimiento
      
      var nombre = row[1];
      var telefono = row[3];
      var fechaNacimiento = new Date(row[4]);
      
      if (!fechaNacimiento || isNaN(fechaNacimiento.getTime())) {
        continue; // Saltar si no hay fecha válida
      }
      
      var diaNac = fechaNacimiento.getDate();
      var mesNac = fechaNacimiento.getMonth() + 1;
      
      // Calcular edad
      var edad = hoy.getFullYear() - fechaNacimiento.getFullYear();
      if (mesHoy < mesNac || (mesHoy === mesNac && diaHoy < diaNac)) {
        edad--;
      }
      
      var socio = {
        nombre: nombre,
        telefono: telefono,
        edad: edad + 1, // Edad que cumplirá
        diaNac: diaNac,
        mesNac: mesNac
      };
      
      // Cumpleaños HOY
      if (diaNac === diaHoy && mesNac === mesHoy) {
        cumpleañosHoy.push(socio);
      }
      
      // Cumpleaños en 3 días
      var fechaEn3Dias = new Date(hoy);
      fechaEn3Dias.setDate(fechaEn3Dias.getDate() + 3);
      if (diaNac === fechaEn3Dias.getDate() && mesNac === (fechaEn3Dias.getMonth() + 1)) {
        cumpleañosProximos3Dias.push(socio);
      }
      
      // Cumpleaños en 7 días
      var fechaEn7Dias = new Date(hoy);
      fechaEn7Dias.setDate(fechaEn7Dias.getDate() + 7);
      if (diaNac === fechaEn7Dias.getDate() && mesNac === (fechaEn7Dias.getMonth() + 1)) {
        cumpleañosProximos7Dias.push(socio);
      }
    }
    
    // Solo enviar email si hay cumpleaños
    if (cumpleañosHoy.length > 0 || cumpleañosProximos3Dias.length > 0 || cumpleañosProximos7Dias.length > 0) {
      enviarEmailCumpleanos(cumpleañosHoy, cumpleañosProximos3Dias, cumpleañosProximos7Dias, emailDestino);
    }
    
    Logger.log("✅ Recordatorio de cumpleaños procesado correctamente");
    
  } catch (error) {
    Logger.log("❌ Error en recordatorio de cumpleaños: " + error.toString());
  }
}

// ============================================
// CONFIGURAR RECORDATORIO AUTOMÁTICO
// ============================================
function configurarRecordatorioAutomatico() {
  // Eliminar triggers anteriores para evitar duplicados
  var triggers = ScriptApp.getProjectTriggers();
  for (var i = 0; i < triggers.length; i++) {
    if (triggers[i].getHandlerFunction() === 'enviarRecordatoriosCumpleanos') {
      ScriptApp.deleteTrigger(triggers[i]);
    }
  }
  
  // Crear nuevo trigger
  // Se ejecutará todos los días a las 8:00 AM
  ScriptApp.newTrigger('enviarRecordatoriosCumpleanos')
    .timeBased()
    .everyDays(1)
    .atHour(8)
    .create();
  
  Logger.log("✅ Recordatorio automático configurado para las 8:00 AM diarias");
  
  // Enviar email de confirmación
  var email = "tu-email@gmail.com"; // ← CAMBIA ESTO
  MailApp.sendEmail({
    to: email,
    subject: "✅ Recordatorios de Cumpleaños Activados - Almíbar",
    body: "El sistema de recordatorios automáticos de cumpleaños ha sido configurado exitosamente.\n\nRecibirás un email cada día a las 8:00 AM con:\n- Cumpleaños del día\n- Cumpleaños en 3 días\n- Cumpleaños en 7 días\n\nSi no hay cumpleaños, no recibirás email.\n\n¡Saludos!\nSistema Automático Almíbar"
  });
}

================================================================================
PERSONALIZAR HORARIO
================================================================================

Si quieres cambiar la hora del recordatorio:

1. En Apps Script, busca esta línea:
   .atHour(8)
   
2. Cambia el número (0-23):
   .atHour(7)   → 7:00 AM
   .atHour(9)   → 9:00 AM
   .atHour(18)  → 6:00 PM

3. Ejecuta nuevamente: configurarRecordatorioAutomatico


================================================================================
DESACTIVAR RECORDATORIOS
================================================================================

Si quieres apagar los recordatorios:

1. Ve a Apps Script
2. Click en el ícono del reloj ⏰ (Activadores)
3. Click en los 3 puntos ⋮ del activador
4. Click en "Eliminar activador"


================================================================================
SOLUCIÓN DE PROBLEMAS
================================================================================

No recibo emails:
→ Verifica que cambiaste el email en el código
→ Revisa tu carpeta de Spam
→ Ejecuta manualmente la función para probar

Error "Authorization required":
→ No autorizaste los permisos
→ Ejecuta configurarRecordatorioAutomatico y autoriza

No hay nadie en "CUMPLEAÑOS HOY":
→ Es normal si nadie cumple años hoy
→ Agrega una fecha de prueba en el Sheet para probar

El activador no aparece:
→ Ejecuta configurarRecordatorioAutomatico de nuevo
→ Autoriza todos los permisos que pida


================================================================================
IMPORTANTE - FORMATO DE FECHAS EN GOOGLE SHEETS
================================================================================

Asegúrate de que en el Google Sheet del Club de Amigos:

- La columna E (Fecha de Nacimiento) esté en formato de FECHA
- No como texto
- Formato recomendado: dd/mm/yyyy (Ej: 15/03/1990)

Para cambiar el formato:
1. Selecciona la columna E
2. Formato → Número → Fecha
